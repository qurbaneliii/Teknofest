for df in [master, kanser, pah, cftr]:

C. FEATURE ENGINEERING — full function
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
def engineer_features(df, flag_cols, fit_cat1_encoder=None):
"""
df: input dataframe
flag_cols: pre-computed list of binary AL columns
fit_cat1_encoder: None for training (fit here), encoder for test/val
Returns: (X_df, cat1_encoder)
"""
d = df.copy()

--- PAGE 3 ---
# ── BLOCK 1: Missingness indicators (BEFORE any imputation) ───────────────
d["miss_AL1_6"] = d["AL_1"].isna().astype(float) # r=0.346 with label
d["miss_AL7_15"] = d["AL_7"].isna().astype(float)
d["miss_AL16_25"] = d["AL_16"].isna().astype(float)
d["miss_AL27_38"] = d["AL_27"].isna().astype(float) # r=0.310 with label
d["miss_EK3"] = d["EK_3"].isna().astype(float) # encodes variant type
# Missingness PCA (captures "which database analyzed this variant")
miss_matrix = d[al_cols].isna().astype(float)
# Note: fit PCA on training data only; transform test data
# Append miss_pc1..miss_pc10 as features
# ── BLOCK 2: Population AF aggregates
─────────────────────────────────────
d["n_pops"] = d[al_raw].notna().sum(axis=1).astype(float)
# n_pops=0 → 82.4% path; n_pops=10 → 84.9%; n_pops=26 → 27.4% (PM2 encoding)
d["n_nonmiss_AL"] = d[al_cols].notna().sum(axis=1).astype(float)
d["max_AF"] = d[al_raw].max(axis=1).fillna(0.0)
d["min_AF_nz"] = d[al_raw].replace(0, np.nan).min(axis=1) # smallest non-zero
d["log_max_AF"] = np.log10(d["max_AF"] + 1e-15)
d["log_min_AF"] = np.log10(d["min_AF_nz"].fillna(1e-15) + 1e-15)
d["BA1_flag"] = (d["max_AF"] > 0.05).astype(float) # ACMG BA1
d["BS1_flag"] = (d["max_AF"] > 0.01).astype(float) # ACMG BS1
d["PM2_flag"] = (d["n_pops"] == 0).astype(float) # ACMG PM2
d["BS2_proxy"] = (d["n_pops"] >= 20).astype(float) # present in many pops
d["n_pops_squared"]= d["n_pops"] ** 2 # captures nonlinearity
# n_pops discrete tier encoding (top-k most informative tiers)
d["npops_tier_0"] = (d["n_pops"] == 0).astype(float)
d["npops_tier_10"] = (d["n_pops"] == 10).astype(float)
d["npops_tier_26"] = (d["n_pops"] == 26).astype(float)
# ── BLOCK 3: Triplet binary flag aggregates
────────────────────────────────
d["sum_flags"] = d[flag_cols].sum(axis=1)
d["n_valid_flags"] = d[flag_cols].notna().sum(axis=1).clip(lower=1)
d["frac_flags"] = d["sum_flags"] / d["n_valid_flags"]
# ── BLOCK 4: EK score engineering
─────────────────────────────────────────
# DO NOT absolute-value any EK column. Negative = benign direction.
# DO NOT impute EK_3 — let GBM handle NaN natively.
d["EK7xEK9"] = d["EK_7"] * d["EK_9"] # AUC boost confirmed
d["EK2xEK4"] = d["EK_2"] * d["EK_4"] # CADD × REVEL

--- PAGE 4 ---
d["EK4xEK6"] = d["EK_4"] * d["EK_6"] # REVEL × AlphaMissense
d["EK7_minus_EK9"] = d["EK_7"] - d["EK_9"] # conservation disagreement
d["max_prob_EK"] = d[["EK_4","EK_5","EK_6"]].max(axis=1) # best prob score
# ACMG PP3/BP4 evidence counts
d["n_EK_path"] = (
(d["EK_2"] > 0).astype(float) +
(d["EK_7"] > 5.0).astype(float) +
(d["EK_9"] > 7.0).astype(float) +
(d["EK_4"] > 0.7).astype(float) +
(d["EK_6"] > 0.7).astype(float) +
(d["EK_3"] > 2.0).astype(float)
)
d["n_EK_ben"] = (
(d["EK_2"] < 0).astype(float) +
(d["EK_7"] < 1.0).astype(float) +
(d["EK_9"] < 3.0).astype(float) +
(d["EK_4"] < 0.3).astype(float) +
(d["EK_6"] < 0.3).astype(float)
)
d["EK_net_evidence"] = d["n_EK_path"] - d["n_EK_ben"] # PP3 minus BP4
# ── BLOCK 5: Categorical features
─────────────────────────────────────────
# CAT_1 decomposition
d["cat1_multipop"] = d["CAT_1"].str.contains("&", na=False).astype(float)
# 95.1% pathogenic — single strongest binary flag in this group
d["cat1_AFR"] = d["CAT_1"].str.contains("AFR", na=False).astype(float)
# 18-30% pathogenic — dominant benign signal
d["cat1_NFE"] = d["CAT_1"].str.contains("NFE", na=False).astype(float)
# 83.3% pathogenic — dominant pathogenic signal
d["cat1_gnomADe"] = d["CAT_1"].str.contains("gnomADe", na=False).astype(float)
d["cat1_gnomADg"] = d["CAT_1"].str.contains("gnomADg", na=False).astype(float)
# CAT_1 target encoding (5-fold, Laplace-smoothed, α=30)
if fit_cat1_encoder is not None:
d["cat1_te"] = d["CAT_1"].map(fit_cat1_encoder).fillna(
fit_cat1_encoder.get("__global__", 0.73))
else:
global_mean = d["Label"].mean() if "Label" in d.columns else 0.73
counts = d.groupby("CAT_1")["Label"].agg(["sum","count"])
alpha = 30
cat1_enc = ((counts["sum"] + alpha * global_mean) / (counts["count"] + alpha)).to_dict()
cat1_enc["__global__"] = global_mean

--- PAGE 5 ---
d["cat1_te"] = d["CAT_1"].map(cat1_enc).fillna(global_mean)
fit_cat1_encoder = cat1_enc
# CAT_2: one-hot (7 AllofUs populations)
cat2_dummies = pd.get_dummies(d["CAT_2"].fillna("missing"), prefix="cat2")
d = pd.concat([d, cat2_dummies], axis=1)
# CAT_6: quality filter
d["cat6_has_filter"] = d["CAT_6"].notna().astype(float)
d["cat6_segdup"] = (d["CAT_6"] == "segdup").astype(float)
d["cat6_lcr"] = (d["CAT_6"] == "lcr").astype(float)
# CAT_3/4/5 genotype consistency
d["geno_consistent"] = (
(d["CAT_3"] == d["CAT_4"]) & (d["CAT_4"] == d["CAT_5"])
).astype(float)
d["geno_missing"] = (d["CAT_3"] == "./.").astype(float)
# ── BLOCK 6: Amino acid change features
───────────────────────────────────
hydrophobic = set("VILMFYWC")
pos_charged = set("RKH")
neg_charged = set("DE")
special = set("GP") # glycine and proline
def aa_class(aa):
if pd.isna(aa): return "unknown"
if aa in hydrophobic: return "hydrophobic"
if aa in pos_charged: return "pos_charged"
if aa in neg_charged: return "neg_charged"
if aa in special: return "special"
return "polar"
d["aa1_class"] = d["AA_1"].apply(aa_class)
d["aa2_class"] = d["AA_2"].apply(aa_class)
d["aa_class_change"] = d["aa1_class"] + "_to_" + d["aa2_class"]
# High-value binary flags (empirically derived from physicochemical analysis)
d["aa_involves_G"] = ((d["AA_1"]=="G") | (d["AA_2"]=="G")).astype(float)
d["aa_involves_P"] = ((d["AA_1"]=="P") | (d["AA_2"]=="P")).astype(float)
d["aa_involves_C"] = ((d["AA_1"]=="C") | (d["AA_2"]=="C")).astype(float)
d["aa_involves_R"] = ((d["AA_1"]=="R") | (d["AA_2"]=="R")).astype(float)
d["aa_to_special"] = (d["aa2_class"] == "special").astype(float)
d["aa_from_special"]= (d["aa1_class"] == "special").astype(float)

--- PAGE 6 ---
d["aa_class_changed"]=(d["aa1_class"] != d["aa2_class"]).astype(float)
# BLOSUM62 (partial — most discriminative substitutions)
BLOSUM62_key = {
("G","R"):-2, ("G","P"):0, ("C","R"):-3, ("C","Y"):-2,
("L","P"):-3, ("R","W"):-3, ("R","Q"): 1, ("E","K"): 1,
("A","T"): 0, ("V","M"): 1, ("S","T"): 1, ("N","D"): 1
}
def get_blosum(r):
key = (str(r["AA_1"]), str(r["AA_2"]))
return BLOSUM62_key.get(key, BLOSUM62_key.get(key[::-1], 0))
d["blosum62_approx"] = d.apply(get_blosum, axis=1)
# For full BLOSUM62: use biopython Bio.Align.substitution_matrices.load("BLOSUM62")
# AA change target encoding
aa_change_str = d["AA_1"].astype(str) + d["AA_2"].astype(str)
if "Label" in d.columns:
global_mean = d["Label"].mean()
counts = d.groupby(aa_change_str)["Label"].agg(["sum","count"])
alpha = 10
aa_enc = ((counts["sum"] + alpha * global_mean) / (counts["count"] + alpha)).to_dict()
d["aa_change_te"] = aa_change_str.map(aa_enc).fillna(global_mean)
return d, fit_cat1_encoder
